// State variables for the non-blocking player tracking
int idx1 = 0, idx2 = 0, idx3 = 0;
unsigned long nextNoteTime1 = 0;
unsigned long nextNoteTime2 = 0;
unsigned long nextNoteTime3 = 0;

void setup() {
  pinMode(pinBuzzer1, OUTPUT);
  pinMode(pinBuzzer2, OUTPUT);
  pinMode(pinBuzzer3, OUTPUT);
}

void loop() {
  unsigned long currentMillis = millis();

  // --- CHANNEL 1 ENGINE (LEAD) ---
  if (currentMillis >= nextNoteTime1 && idx1 < (sizeof(fettyLead) / sizeof(int))) {
    int note = pgm_read_word(&fettyLead[idx1]);
    int durationType = pgm_read_word(&fettyLead[idx1 + 1]);
    long duration = wholeNoteDuration / durationType;
    
    if (note == REST) {
      noTone(pinBuzzer1);
    } else {
      tone(pinBuzzer1, note, duration * 0.9); // 10% space between notes
    }
    nextNoteTime1 = currentMillis + duration;
    idx1 += 2;
  }

  // --- CHANNEL 2 ENGINE (ARP) ---
  if (currentMillis >= nextNoteTime2 && idx2 < (sizeof(fettyArp) / sizeof(int))) {
    int note = pgm_read_word(&fettyArp[idx2]);
    int durationType = pgm_read_word(&fettyArp[idx2 + 1]);
    long duration = wholeNoteDuration / durationType;
    
    if (note == REST) {
      noTone(pinBuzzer2);
    } else {
      tone(pinBuzzer2, note, duration * 0.9);
    }
    nextNoteTime2 = currentMillis + duration;
    idx2 += 2;
  }

  // --- CHANNEL 3 ENGINE (BASS) ---
  if (currentMillis >= nextNoteTime3 && idx3 < (sizeof(fettyBass) / sizeof(int))) {
    int note = pgm_read_word(&fettyBass[idx3]);
    int durationType = pgm_read_word(&fettyBass[idx3 + 1]);
    long duration = wholeNoteDuration / durationType;
    
    if (note == REST) {
      noTone(pinBuzzer3);
    } else {
      tone(pinBuzzer3, note, duration * 0.9);
    }
    nextNoteTime3 = currentMillis + duration;
    idx3 += 2;
  }

  // Optional: Reset song at the end to loop it infinitely
  if (idx1 >= (sizeof(fettyLead) / sizeof(int)) &&
      idx2 >= (sizeof(fettyArp) / sizeof(int)) &&
      idx3 >= (sizeof(fettyBass) / sizeof(int))) {
    delay(2000); // Small pause at the absolute end of the song
    idx1 = 0; idx2 = 0; idx3 = 0;
    nextNoteTime1 = millis();
    nextNoteTime2 = millis();
    nextNoteTime3 = millis();
  }
}
